// CPU Scheduling 計算 waiting time 的簡化偽程式
// processes: 每個 process 有 name, arrival, burst
function fcfs(processes) {
    let time = 0;
    let result = [];
    processes.sort((a, b) => a.arrival - b.arrival);
    for (const p of processes) {
        if (time < p.arrival) time = p.arrival; // CPU idle
        const start = time;
        const finish = start + p.burst;
        const waiting = start - p.arrival;
        result.push({ name: p.name, start, finish, waiting });
        time = finish;
    }
    return result;
}

deadlock、resource-allocation graph、cycle。若是計算題，請特別注意 arrival time、burst time、waiting time、turnaround time、ready queue、time quantum、cycle 等名詞；若是圖形題，請標示箭頭方向與每個節點意義。

11、時間複雜度與考試容易錯的地方

時間複雜度：觀念題通常不需計算；排程或圖形判斷若逐一檢查 n 個 process，基本整理成本約 O(n) 到 O(n log n)，依排序方式而定。

容易錯：不要只背名詞，要能畫出對應圖形；CPU scheduling 要分清楚 waiting time 與 turnaround time；deadlock 要看 cycle 與 resource instance 數量。

12、Deadlock 判斷概念程式碼與複製鍵

// Deadlock 判斷概念：若資源配置圖形成 cycle，且每種資源只有一個 instance，則 deadlock
// 這裡用概念性偽程式表示
function hasDeadlock(graph) {
    // graph 包含 process 與 resource 的有向邊
    // request edge: P -> R
    // assignment edge: R -> P
    return hasCycle(graph); // 若有 cycle，可能代表 deadlock
}

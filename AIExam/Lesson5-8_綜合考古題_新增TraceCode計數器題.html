考試重點整理

最像老師會出成大題的是：Linked List pointer 修改、Binary Tree Traversal、BST 建立與刪除、Max Heap 建立與刪除、Infix to Postfix、Postfix Evaluation、CPU Scheduling、Deadlock Resource-Allocation Graph。

老師題型通常「題幹短、作答多」：看起來只有一句話，但需要畫圖、追蹤 stack 或 pointer、列出每一步。

Trace code 題型整理

Trace code 的解題順序：先畫資料結構初始狀態，再逐行看指標或變數如何改變，最後才寫輸出。Linked list 追蹤 next；tree 追蹤 left/right 和遞迴順序；stack 追蹤 push/pop；scheduling 追蹤 ready queue 和 Gantt chart。

題目原文
Given the following singly linked list: A → C → F → H → null a. Insert D between C and F. b. Delete H from the list. c. Draw the linked list after each operation.
中文翻譯與核心觀念

給一個單向鏈結串列，先把 D 插入 C 和 F 中間，再刪除 H。考 linked list 的 node、next pointer、head、null、insert、delete。

解題步驟
  1. 原始串列：Head → A → C → F → H → null。
  2. 插入 D：找到 C，讓 D.next 指向 F，再讓 C.next 指向 D。
  3. 刪除 H：找到 H 的前一個節點 F，讓 F.next 指向 null。

動畫步驟

完整答案：插入後 A → C → D → F → H → null；刪除 H 後 A → C → D → F → null。
考試容易錯：插入時一定要先接新節點到後方，再讓前一節點指向新節點，否則可能遺失後半段串列。

Trace Code 計數器題解題技巧

追蹤計數器題時，先不要急著算最後答案。建議先畫資料結構狀態，再建立「每一次進入迴圈或遞迴」的表格。遇到 linked list 要確認哪些節點被 pointer 跳過；遇到 tree 要寫出真正拜訪順序；遇到 BST 要把每一次節點比較都算進 counter；遇到 heap 要分清楚 compare 與 swap；遇到 stack 要記得 operator 會 pop 兩次，且計算結果 push 回去也算一次 push。

題目原文
Write the output of the following C program. What is the final value of count?
中文翻譯與核心觀念

請追蹤 linked list 程式。程式會走訪節點並依照資料值增加 count，中途還會修改 pointer 跳過一個節點。考點是 linked list traversal、pointer tracing、條件計數。

完整 C 程式碼
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;              // 節點資料
    struct Node *next;     // 指向下一個節點
};

void printList(struct Node *head) {
    while (head != NULL) {
        printf("%d ", head->data);
        head = head->next;
    }
    printf("NULL\n");
}

int main(void) {
    struct Node n1, n2, n3, n4, n5, n6;
    struct Node *head = &n1;
    struct Node *p;
    int count = 0;

    n1.data = 1; n2.data = 4; n3.data = 7;
    n4.data = 2; n5.data = 9; n6.data = 6;

    n1.next = &n2;
    n2.next = &n3;
    n3.next = &n4;
    n4.next = &n5;
    n5.next = &n6;
    n6.next = NULL;

    p = head;
    while (p != NULL) {
        if (p->data % 2 == 0)
            count += 2;      // 偶數加 2
        else
            count += 1;      // 奇數加 1

        if (p->data > 5)
            count += 3;      // 大於 5 再加 3

        if (p->data == 4 && p->next != NULL) {
            p->next = p->next->next; // 跳過 7
            count += 4;               // 修改指標加 4
        }

        p = p->next;
    }

    printList(head);
    printf("count = %d\n", count);
    return 0;
}

#include <stdio.h>

struct Node {
    char data;
    struct Node *left;
    struct Node *right;

    return x != NULL && x->left == NULL && x->right == NULL;

        inorder(root->left);

            count += 2;            // leaf 加 2

        if (root->left != NULL && root->right != NULL)
            count += 3;            // 有兩個 children 加 3

        inorder(root->right);
    }
}

題目原文
Write the output of the following C program. What are the final values of counter, found, miss, and total?

核心觀念：BST search 中，每比較一個節點 counter 加 1；找到 found 加 1；找不到 miss 加 2。

#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *left;
    struct Node *right;
};

int counter = 0;
int found = 0;
int miss = 0;

struct Node* newNode(int x) {
    struct Node* n = (struct Node*)malloc(sizeof(struct Node));
    n->data = x;
    n->left = n->right = NULL;
    return n;
}

struct Node* insert(struct Node* root, int x) {
    if (root == NULL) return newNode(x);
    if (x < root->data) root->left = insert(root->left, x);
    else root->right = insert(root->right, x);
    return root;
}

void search(struct Node* root, int key) {
    struct Node* p = root;
    while (p != NULL) {
        counter++;                 // 每比較一個節點就加 1
        if (key == p->data) {
            found++;               // 找到 key
            return;
        } else if (key < p->data) {
            p = p->left;
        } else {
            p = p->right;
        }
    }
    miss += 2;                     // 搜尋失敗加 2
}

int main(void) {
    int a[] = {9, 5, 12, 3, 7, 10, 15, 6, 8};
    int i;
    struct Node* root = NULL;

    for (i = 0; i < 9; i++)
        root = insert(root, a[i]);

    search(root, 6);
    search(root, 13);
    search(root, 8);

    printf("counter = %d\n", counter);
    printf("found = %d\n", found);
    printf("miss = %d\n", miss);
    printf("total = %d\n", counter + found + miss);
    return 0;
}

題目原文
Write the output of the following C program. What are the final values of compare, swaps, and count?

核心觀念：刪除 max heap 的 root 後，把最後一個元素放到 root，再 heapify down。每比較一次 child，compare 加 1；每交換一次，swaps 加 1。

#include <stdio.h>

int compare = 0;
int swaps = 0;

void swap(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
    swaps++;                 // 每交換一次加 1
}

void heapifyDown(int heap[], int n, int i) {
    while (1) {
        int largest = i;
        int left = 2 * i + 1;
        int right = 2 * i + 2;

        if (left < n) {
            compare++;       // 比較 left child
            if (heap[left] > heap[largest])
                largest = left;
        }

        if (right < n) {
            compare++;       // 比較 right child
            if (heap[right] > heap[largest])
                largest = right;
        }

        if (largest == i)
            break;

        swap(&heap[i], &heap[largest]);
        i = largest;
    }
}

int main(void) {
    int heap[10] = {20, 18, 15, 10, 12, 9, 13, 4, 8, 11};
    int n = 10;
    int i;

    heap[0] = heap[n - 1];   // 最後元素 11 放到 root
    n--;                     // 刪除原本 root 20

    heapifyDown(heap, n, 0);

    for (i = 0; i < n; i++)
        printf("%d ", heap[i]);

    printf("\ncompare = %d\n", compare);
    printf("swaps = %d\n", swaps);
    printf("count = %d\n", compare + swaps);
    return 0;
}

題目原文
Write the output of the following C program. What are the final values of pushCount, popCount, opCount, and totalOps?

核心觀念：postfix evaluation + stack 操作計數。注意 operator 會 pop 兩次，計算結果還會 push 回 stack。

#include <stdio.h>
#include <ctype.h>

int stack[100];
int top = -1;
int pushCount = 0;
int popCount = 0;
int opCount = 0;

void push(int x) {
    stack[++top] = x;
    pushCount++;             // 每次 push 都計數
}

int pop(void) {
    popCount++;              // 每次 pop 都計數
    return stack[top--];
}

int isOperator(char c) {
    return c == '+' || c == '-' || c == '*' || c == '/';
}

int main(void) {
    char exp[] = "5 2 3 * + 8 4 / - 6 +";
    int i = 0;

    while (exp[i] != '\0') {
        if (isdigit(exp[i])) {
            push(exp[i] - '0');
        } else if (isOperator(exp[i])) {
            int b = pop();
            int a = pop();
            int r = 0;
            opCount++;       // 每遇到一個 operator 就計數

            if (exp[i] == '+') r = a + b;
            if (exp[i] == '-') r = a - b;
            if (exp[i] == '*') r = a * b;
            if (exp[i] == '/') r = a / b;

            push(r);         // 計算結果也要 push 回 stack
        }
        i++;
    }

    printf("result = %d\n", pop());
    printf("pushCount = %d\n", pushCount);
    printf("popCount = %d\n", popCount);
    printf("opCount = %d\n", opCount);
    printf("totalOps = %d\n", pushCount + popCount + opCount);
    return 0;
}

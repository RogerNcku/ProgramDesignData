1、題目原文

Convert the following infix expression into postfix and prefix expressions. A + B * C

2、題目中文翻譯

3、題目核心觀念

7、完整 C 程式碼：Infix-to-Postfix

C 程式碼：infix_to_postfix.c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

#define MAX 200

/* 使用陣列實作 stack，專門存放運算子 */
char stack[MAX];
int top = -1;

/* 判斷 stack 是否為空 */
int isEmpty() {
    return top == -1;
}

/* 將運算子放入 stack */
void push(char c) {
    stack[++top] = c;
}

/* 從 stack 取出最上面的運算子 */
char pop() {
    if (isEmpty()) return '\0';
    return stack[top--];
}

/* 查看 stack 最上方的運算子，但不移除 */
char peek() {
    if (isEmpty()) return '\0';
    return stack[top];
}

/* 回傳運算子的優先權 */
int precedence(char op) {
    if (op == '^') return 3;
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;
}

/* 將 infix 轉成 postfix */
void infixToPostfix(const char infix[], char postfix[]) {
    int i = 0;
    postfix[0] = '\0';

    while (infix[i] != '\0') {
        char c = infix[i];

        /* 跳過空白 */
        if (isspace(c)) {
            i++;
            continue;
        }

        /* 如果是 operand，可能是 x1、y2、12 這種多字元資料 */
        if (isalnum(c)) {
            char token[MAX];
            int k = 0;

            while (isalnum(infix[i])) {
                token[k++] = infix[i++];
            }

            token[k] = '\0';
            strcat(postfix, token);
            strcat(postfix, " ");
            continue;
        }

        /* 左括號直接 push */
        if (c == '(') {
            push(c);
        }
        /* 右括號：pop 到遇到左括號為止 */
        else if (c == ')') {
            while (!isEmpty() && peek() != '(') {
                char op[3] = {pop(), ' ', '\0'};
                strcat(postfix, op);
            }
            pop(); // 移除左括號
        }
        /* 一般運算子 */
        else {
            while (!isEmpty() &&
                   peek() != '(' &&
                   precedence(peek()) >= precedence(c)) {
                char op[3] = {pop(), ' ', '\0'};
                strcat(postfix, op);
            }
            push(c);
        }

        i++;
    }

    /* 把 stack 剩下的運算子全部輸出 */
    while (!isEmpty()) {
        char op[3] = {pop(), ' ', '\0'};
        strcat(postfix, op);
    }
}

int main() {
    char infix[] = "x1 + y2 * 12 ^ (x2 ^ x3) / (z + 25)";
    char postfix[MAX];

    infixToPostfix(infix, postfix);

    printf("Infix: %s\n", infix);
    printf("Postfix: %s\n", postfix);

    return 0;
}

9、函式與語法說明

  • push：把 operator 放入 stack。
  • pop：取出 stack top 的 operator。
  • peek：查看 stack top，但不刪除。
  • precedence：回傳 +、-、*、/、^ 的優先權。
  • isalnum：判斷是否為英文或數字，用來讀多字元 operand。
  • strcat：把 token 接到 postfix 字串後面。

10、執行結果

Infix: x1 + y2 * 12 ^ (x2 ^ x3) / (z + 25) Postfix: x1 y2 12 x2 x3 ^ ^ * z 25 + / +

11、時間複雜度與考試提醒

每個 token 最多被讀一次，每個 operator 最多 push/pop 一次，所以時間複雜度為 O(n)。

考試時要注意多字元 operand，例如 x1、12、x3，不能拆成 x 和 1 或 1 和 2。

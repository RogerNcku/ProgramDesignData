原始程式碼

      
#include <stdio.h>

int f(char *s, char *t)
{
    char *p1, *p2;

    for (p1 = s; *p1 != '\0'; p1++)
    {
        for (p2 = t; *p2; p2++)
        {
            if (*p1 == *p2)
            {
                break;
            }
        }

        if (*p2 == '\0')
        {
            break;
        }
    }

    return p1 - s;
}

int main(void)
{
    printf("f(\"abcd\", \"babc\") = %d\n", f("abcd", "babc"));
    printf("f(\"abcd\", \"bcd\") = %d\n", f("abcd", "bcd"));

    return 0;
}


    

    

      
函式說明

      
  • f
    參數：char *s, char *t
    找出字串 s 中第一個不屬於字串 t 的字元位置。
    • 

  • main
    參數：void
    主程式，負責建立測試資料、呼叫函式並印出結果。

      
預期執行結果

      
f("abcd", "babc") = 3
f("abcd", "bcd") = 0

    自訂函式。參數：void。主程式進入點，負責建立測試資料、呼叫函式、顯示結果。

    把格式化文字輸出到螢幕。

原始程式碼

      
#include <stdio.h>
#include <string.h>

void create_txt_extension(char *file_name)
{
    int i;
    int has_extension = 0;

    for (i = 0; file_name[i] != '\0'; i++)
    {
        if (file_name[i] == '.')
        {
            has_extension = 1;
            break;
        }
    }

    if (!has_extension)
    {
        strcat(file_name, ".txt");
    }
}

int main(void)
{
    char file1[50] = "output";
    char file2[50] = "result.txt";

    create_txt_extension(file1);
    create_txt_extension(file2);

    printf("file1 = %s\n", file1);
    printf("file2 = %s\n", file2);

    return 0;
}


    

    

      
函式說明

      
  • create_txt_extension
    參數：char *file_name
    若檔名沒有副檔名，就在字串尾端加上 .txt。
    • 

  • main
    參數：void
    主程式，負責建立測試資料、呼叫函式並印出結果。

practice10_1.c

原始程式碼

#include <stdio.h>
#include <stdlib.h>

void *my_malloc(size_t n)
{
    void *p;

    p = malloc(n);

    if (p == NULL)
    {
        printf("Error: malloc failed.\n");
        exit(1);
    }

    return p;
}

int main(void)
{
    int *arr;
    int i;

    arr = (int *)my_malloc(5 * sizeof(int));

    for (i = 0; i < 5; i++)
    {
        arr[i] = i + 1;
    }

    for (i = 0; i < 5; i++)
    {
        printf("arr[%d] = %d\n", i, arr[i]);
    }

    free(arr);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

int *create_array(int n, int initial_value)
{
    int *arr;
    int i;

    arr = (int *)malloc(n * sizeof(int));

    if (arr == NULL)
    {
        return NULL;
    }

    for (i = 0; i < n; i++)
    {
        arr[i] = initial_value;
    }

    return arr;
}

int main(void)
{
    int *arr;
    int n;
    int value;
    int i;

    printf("Enter array size: ");
    scanf("%d", &n);

    printf("Enter initial value: ");
    scanf("%d", &value);

    arr = create_array(n, value);

    if (arr == NULL)
    {
        printf("Array allocation failed.\n");
        return 1;
    }

    for (i = 0; i < n; i++)
    {
        printf("arr[%d] = %d\n", i, arr[i]);
    }

    free(arr);

    return 0;
}

原始程式碼

      
#include <stdio.h>

double inner_product(const double *a, const double *b, int n)
{
    int i;
    double sum = 0.0;

    for (i = 0; i < n; i++)
    {
        sum += a[i] * b[i];
    }

    return sum;
}

int main(void)
{
    double a[] = {1.0, 2.0, 3.0};
    double b[] = {4.0, 5.0, 6.0};
    double result;

    result = inner_product(a, b, 3);

    printf("Inner product = %.2f\n", result);

    return 0;
}

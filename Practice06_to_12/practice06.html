原始程式碼

      
#include <stdio.h>

void swap(int *p, int *q)
{
    int temp;

    temp = *p;
    *p = *q;
    *q = temp;
}

int main(void)
{
    int i = 10;
    int j = 20;

    printf("Before swap: i = %d, j = %d\n", i, j);

    swap(&i, &j);

    printf("After swap: i = %d, j = %d\n", i, j);

    return 0;
}


    

    

      
函式說明

      
  • swap
    參數：int *p, int *q
    使用指標交換兩個整數變數的值。

    參數：void

#include <stdio.h>

void split_time(long total_sec, int *hr, int *min, int *sec)
{
    *hr = total_sec / 3600;
    total_sec = total_sec % 3600;

    *min = total_sec / 60;
    *sec = total_sec % 60;
}

int main(void)
{
    long total_sec;
    int hr, min, sec;

    printf("Enter total seconds: ");
    scanf("%ld", &total_sec);

    split_time(total_sec, &hr, &min, &sec);

    printf("%02d:%02d:%02d\n", hr, min, sec);

    return 0;
}


    

    

      
函式說明

      
  • split_time
    參數：long total_sec, int *hr, int *min, int *sec
    把總秒數轉換成時、分、秒，並透過指標回傳結果。
    • 

  • main
    參數：void
    主程式，負責建立測試資料、呼叫函式並印出結果。

原始程式碼

      
#include <stdio.h>

void find_two_largest(int a[], int n, int *largest, int *second_largest)
{
    int i;

    if (a[0] > a[1])
    {
        *largest = a[0];
        *second_largest = a[1];
    }
    else
    {
        *largest = a[1];
        *second_largest = a[0];
    }

    for (i = 2; i < n; i++)
    {
        if (a[i] > *largest)
        {
            *second_largest = *largest;
            *largest = a[i];
        }
        else if (a[i] > *second_largest)
        {
            *second_largest = a[i];
        }
    }
}

int main(void)
{
    int a[] = {10, 5, 30, 20, 40, 15};
    int largest, second_largest;
    int n = sizeof(a) / sizeof(a[0]);

    find_two_largest(a, n, &largest, &second_largest);

    printf("Largest = %d\n", largest);
    printf("Second largest = %d\n", second_largest);

    return 0;
}


    

    

      
函式說明

      
  • find_two_largest
    參數：int a[], int n, int *largest, int *second_largest
    在整數陣列中找出最大值與第二大值。
    • 

  • main
    參數：void
    主程式，負責建立測試資料、呼叫函式並印出結果。

      
預期執行結果

      
Largest = 40
Second largest = 30

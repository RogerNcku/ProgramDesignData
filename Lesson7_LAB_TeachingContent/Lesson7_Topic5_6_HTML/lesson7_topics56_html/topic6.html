（一）這張投影片在說什麼？ 這張投影片開始介紹比較正式的 Stack 轉換法。程式由左到右掃描 Infix，遇到 operand 直接輸出，遇到 operator 則依照優先順序和 Stack 頂端比較。

四、完整中文講解

（一）Stack 在這裡的用途

Stack 是後進先出資料結構。中序轉後序時，運算元可以直接輸出，但是運算子不能馬上輸出，因為還要考慮後面是否有更高優先順序的運算。因此程式會把運算子暫時放進 stack，等到適當時機再 pop 出來。

（二）遇到不同符號的規則

運算元 Operand
例如 A、B、C、2、3。直接加入 postfix output。
左括號 (
直接 push 到 stack，表示括號內部要先處理。
右括號 )
一直 pop stack 並輸出，直到遇到左括號；左括號只移除不輸出。
運算子 Operator
比較目前運算子與 stack 頂端運算子的優先順序。若 stack 頂端較高或相同，先 pop 輸出，再把目前運算子 push。

（三）為什麼這樣可以轉成 Postfix？

Postfix 的特徵是運算子出現在它的兩個運算元之後。使用 stack 時，運算元先輸出，運算子先暫存，等到它確定應該被計算時再輸出。因此最後 output 的順序自然會變成「運算元、運算元、運算子」的後序形式。

小範例：
Infix：A + B * C
讀 A → output = A
讀 + → push +
讀 B → output = AB
讀 * → * 優先於 +，push *
讀 C → output = ABC
輸入結束 → pop *、pop +
Postfix：ABC*+

程式題目教學內容

一、程式題目講解

這個程式不只完成中序轉後序，還會印出每一步的 Stack 與 Output，讓學生可以對照演算法表格理解每個字元進入時發生的動作。

二、完整程式碼

以下保留本頁原本的完整 C 程式碼，並搭配中文註解說明主要功能與流程。

#include <stdio.h>
#include <ctype.h>
#include <string.h>

/*
    主題 6：Infix-to-Postfix using a Stack
    說明：
    這份程式不只轉換 postfix，也會印出每一步的 Stack 與 Output，
    對應投影片 9-11 的表格示範。

    範例輸入：A+B*C^(2^3)/(A+B)
*/

#define MAX 300

char stack[MAX];
int top = -1;

void push(char ch) {
    if (top < MAX - 1) stack[++top] = ch;
}

char pop(void) {
    if (top >= 0) return stack[top--];
    return '\0';
}

char peek(void) {
    if (top >= 0) return stack[top];
    return '\0';
}

int isEmpty(void) {
    return top == -1;
}

int priority(char op) {
    if (op == '^') return 3;
    if (op == '*' || op == '/') return 2;
    if (op == '+' || op == '-') return 1;
    return 0;
}

int isOperator(char ch) {
    return ch == '+' || ch == '-' || ch == '*' || ch == '/' || ch == '^';
}

void getStackString(char result[]) {
    int i, j = 0;
    if (isEmpty()) {
        strcpy(result, "Empty");
        return;
    }
    for (i = 0; i <= top; i++) {
        result[j++] = stack[i];
        result[j++] = ' ';
    }
    result[j] = '\0';
}

void printStep(char x, const char action[], const char output[]) {
    char stackText[MAX];
    getStackString(stackText);
    if (x == '\0') {
        printf("%-8s | %-18s | %-22s | %s\n", "End", stackText, output, action);
    } else {
        printf("%-8c | %-18s | %-22s | %s\n", x, stackText, output, action);
    }
}

int main(void) {
    char infix[MAX];
    char output[MAX] = "";
    int i;

    printf("請輸入中序式，例如 A+B*C^(2^3)/(A+B)：");
    scanf("%299s", infix);

    printf("\n%-8s | %-18s | %-22s | %s\n", "X", "Stack S", "Output", "說明");
    printf("--------------------------------------------------------------------------------\n");

    for (i = 0; infix[i] != '\0'; i++) {
        char x = infix[i];
        char temp[3] = {x, '\0', '\0'};

        if (isalnum(x)) {
            // 運算元直接輸出
            strcat(output, temp);
            printStep(x, "運算元直接加入 output", output);
        }
        else if (x == '(') {
            // 左括號直接 push
            push(x);
            printStep(x, "左括號 push 到 stack", output);
        }
        else if (x == ')') {
            // 右括號：pop 到遇到左括號
            while (!isEmpty() && peek() != '(') {
                char op[2] = {pop(), '\0'};
                strcat(output, op);
            }
            pop(); // 移除左括號
            printStep(x, "右括號：pop 到遇到左括號並丟掉左括號", output);
        }
        else if (isOperator(x)) {
            // 運算子：先 pop 較高或相同優先順序的運算子
            while (!isEmpty() && peek() != '(' && priority(peek()) >= priority(x)) {
                char op[2] = {pop(), '\0'};
                strcat(output, op);
            }
            push(x);
            printStep(x, "運算子：比較優先順序後 push", output);
        }
    }

    // 最後把 stack 中所有運算子全部 pop 出來
    while (!isEmpty()) {
        char op[2] = {pop(), '\0'};
        strcat(output, op);
    }
    printStep('\0', "輸入結束：清空 stack", output);

    printf("\nPostfix 結果：%s\n", output);
    return 0;
}

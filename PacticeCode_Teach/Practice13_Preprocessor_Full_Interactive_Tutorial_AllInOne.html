預處理器不是在「執行程式」，而是在編譯前先改寫原始碼，例如把巨集替換掉、把標頭檔內容插入程式，再交給 compiler。

二、完整投影片式教學

Slide 1：Preprocessor 是什麼？

Preprocessor 是編譯前先執行的處理階段。凡是以 # 開頭的 directive，例如 #include、#define、#ifdef，都會先被預處理器處理。

  • #define：巨集定義。
  • #include：檔案包含。
  • #ifdef / #if / #endif：條件編譯。

Slide 2：Simple Macro

#define STR_LEN 20 會讓程式中所有 STR_LEN 在預處理後被替換成 20。

#define STR_LEN 20
char str[STR_LEN];   // 預處理後變成 char str[20];

Slide 3：Parameterized Macro

參數巨集看起來像函式，但它不是函式呼叫，而是文字替換。

#define SQUARE(a) a * a
SQUARE(3+6)  // 展開成 3+6*3+6，結果是 27，不是 81

使用參數巨集時，參數與整個 replacement list 都要盡量加括號，避免運算優先順序錯誤。

Slide 4：巨集的安全寫法

#define SQUARE(a) ((a) * (a))
#define MAX(x, y) (((x) > (y)) ? (x) : (y))

這樣 SQUARE(3+6) 會展開成 ((3+6) * (3+6))，結果才會是 81。

Slide 5：File Inclusion

#include <stdio.h> 通常用於系統標頭檔；#include "average.h" 通常用於自己寫的標頭檔。

寫法用途
#include <filename>標準函式庫或系統提供的 header。
#include "filename"自己專案中的 header，例如 average.h。

Slide 6：Header Guard

標頭檔常使用 header guard 避免重複 include。

#ifndef PRACTICE13_3_AVERAGE_H
#define PRACTICE13_3_AVERAGE_H
double calculate_average(int scores[3]);
#endif

Slide 7：Conditional Compilation

條件編譯可控制某些程式碼是否被編譯，例如 debug 訊息。

#define DEBUG 1
#if DEBUG
printf("debug message\n");
#endif

四、第 1 題：撰寫參數巨集

題目

請寫出可以計算下列功能的 parameter macros：

  1. 求 x 和 y 的最大值。
  2. 求陣列大小。
  3. 求兩個數字平均。
  4. 判斷某數是否為偶數。

解題觀念

本題的重點是「參數一定要加括號」。例如 MAX(a+b, c) 若沒有括號，很容易因運算優先順序造成錯誤。ARRAY_SIZE(arr) 使用 sizeof(arr) / sizeof(arr[0])，但只能用在真正陣列，不能用在指標。

程式

practice13_1_2_macros_demo.c

/*
 * ============================================================
 * Practice 13 第 1、2 題：參數巨集 parameter macros
 * ============================================================
 *
 * 第 1 題：請寫出可以計算下列功能的參數巨集：
 *   1-1. MAX(x, y)        ：求 x 和 y 的最大值
 *   1-2. ARRAY_SIZE(arr)  ：求陣列大小
 *   1-3. AVG(x, y)        ：求兩個數字的平均
 *   1-4. IS_EVEN(n)       ：判斷數字是否為偶數
 *
 * 第 2 題：#define T(x) (x*10)
 *   2-1. T(3+6) 的結果是多少？
 *   2-2. 若想讓結果變成 90，應如何修正？
 */

#include <stdio.h>   /* 第 0 節：引入 printf() 所需要的標頭檔 */

/*
 * 第 1-1 題：MAX(x, y)
 * 功能：回傳 x 和 y 中較大的值。
 * 說明：每個參數都加上括號，可避免運算優先順序造成錯誤。
 */
#define MAX(x, y) (((x) > (y)) ? (x) : (y))

/*
 * 第 1-2 題：ARRAY_SIZE(arr)
 * 功能：計算陣列元素個數。
 * 原理：整個陣列的 byte 數 / 單一元素的 byte 數。
 * 注意：這個巨集只能用在真正的陣列，不能用在指標。
 */
#define ARRAY_SIZE(arr) (sizeof(arr) / sizeof((arr)[0]))

/*
 * 第 1-3 題：AVG(x, y)
 * 功能：計算兩個數字的平均值。
 * 說明：除以 2.0 可以保留小數，避免整數除法只留下整數。
 */
#define AVG(x, y) (((x) + (y)) / 2.0)

/*
 * 第 1-4 題：IS_EVEN(n)
 * 功能：判斷 n 是否為偶數。
 * 回傳：若 n 可以被 2 整除，結果為 1；否則為 0。
 */
#define IS_EVEN(n) (((n) % 2) == 0)

/*
 * 第 2-1 題：錯誤示範
 * T_BAD(3+6) 會展開成 (3+6*10)，因為乘法優先於加法，結果為 63。
 */
#define T_BAD(x) (x*10)

/*
 * 第 2-2 題：正確寫法
 * 將參數 x 加上括號，T_GOOD(3+6) 會展開成 ((3+6)*10)，結果為 90。
 */
#define T_GOOD(x) ((x)*10)

int main(void)
{
    /* 第 3 節：宣告測試資料 */
    int a = 10;
    int b = 25;
    int numbers[] = {3, 6, 9, 12, 15};

    /* 第 4 節：依序輸出每一題的測試結果 */
    printf("Practice 13 第 1、2 題：Macro Demo\n");
    printf("第 1-1 題 MAX(%d, %d) = %d\n", a, b, MAX(a, b));
    printf("第 1-2 題 ARRAY_SIZE(numbers) = %lu\n", (unsigned long)ARRAY_SIZE(numbers));
    printf("第 1-3 題 AVG(%d, %d) = %.2f\n", a, b, AVG(a, b));
    printf("第 1-4 題 IS_EVEN(%d) = %s\n", a, IS_EVEN(a) ? "true" : "false");
    printf("第 1-4 題 IS_EVEN(%d) = %s\n", b, IS_EVEN(b) ? "true" : "false");
    printf("第 2-1 題 T_BAD(3+6) = %d\n", T_BAD(3+6));
    printf("第 2-2 題 T_GOOD(3+6) = %d\n", T_GOOD(3+6));

    return 0;
}

執行結果

Practice 13 第 1、2 題：Macro Demo 第 1-1 題 MAX(10, 25) = 25 第 1-2 題 ARRAY_SIZE(numbers) = 5 第 1-3 題 AVG(10, 25) = 17.50 第 1-4 題 IS_EVEN(10) = true 第 1-4 題 IS_EVEN(25) = false 第 2-1 題 T_BAD(3+6) = 63 第 2-2 題 T_GOOD(3+6) = 90

程式說明

巨集功能重點
MAX(x,y)回傳較大值使用三元運算子 ?:。
ARRAY_SIZE(arr)計算陣列元素個數整個陣列 byte 數除以單一元素 byte 數。
AVG(x,y)計算平均除以 2.0 保留小數。
IS_EVEN(n)判斷偶數((n)%2)==0 為真代表偶數。

五、第 2 題：T(3+6) 的巨集展開

題目

#define T(x) (x*10)

  1. T(3+6) 的值是多少？
  2. 如果想讓結果是 90，要怎麼修正？

答案

T(3+6) 會展開成 (3+6*10)，因為乘法優先於加法，所以結果是 63。

要得到 90，應改成 #define T(x) ((x)*10)，讓 3+6 先被括號包住。

程式

第 2 題已整合在 practice13_1_2_macros_demo.c 中，使用 T_BAD 與 T_GOOD 對照。

#define T_BAD(x) (x*10)
#define T_GOOD(x) ((x)*10)

printf("T_BAD(3+6) = %d\n", T_BAD(3+6));
printf("T_GOOD(3+6) = %d\n", T_GOOD(3+6));

程式結果

第 2-1 題 T_BAD(3+6) = 63 第 2-2 題 T_GOOD(3+6) = 90

程式說明

T_BAD 的錯誤不是 compiler 算錯，而是 macro 只是文字替換。當你寫 T_BAD(3+6)，預處理器只把 x 換成 3+6，所以變成 (3+6*10)。因此巨集參數要寫成 (x)。

六、第 3 題：多檔案 average.h / average.c / main.c

題目

請寫一個計算三科期中考平均分數的程式：

  1. 在 average.h 宣告函式。
  2. 在 average.c 定義函式，接收大小為 3 的陣列並回傳平均。
  3. 在 main.c 使用此函式，讀入使用者指定的三科成績並印出平均。

解題觀念

.h 放函式宣告，.c 放函式定義，main.c 負責主流程。編譯時要把 main.c 和 average.c 一起交給 gcc。

程式 1：practice13_3_average.h

/*
 * ============================================================
 * Practice 13 第 3-1 題：average.h
 * ============================================================
 *
 * 本檔案負責「宣告」計算三科平均分數的函式。
 * 題目要求：Declare the function in a file "average.h"。
 */

#ifndef PRACTICE13_3_AVERAGE_H
#define PRACTICE13_3_AVERAGE_H

/*
 * 第 3-1 題：calculate_average 函式宣告
 * 參數：scores[3]，代表三科成績的整數陣列。
 * 回傳：三科成績的平均值，型態為 double。
 */
double calculate_average(int scores[3]);

#endif

程式 2：practice13_3_average.c

/*
 * ============================================================
 * Practice 13 第 3-1 題：average.c
 * ============================================================
 *
 * 本檔案負責「定義」calculate_average() 函式。
 * 題目要求：Define it in a file "average.c"。
 */

#include "practice13_3_average.h"  /* 第 0 節：引入函式宣告，確保宣告與定義一致 */

/*
 * 第 3-1 題：calculate_average 函式定義
 * 功能：接收一個大小為 3 的陣列，計算三個數字的平均。
 * 為什麼回傳 double：平均可能有小數，所以不能只用 int。
 */
double calculate_average(int scores[3])
{
    int sum = 0;  /* 第 1 節：用來累加三科成績 */
    int i;        /* 第 1 節：迴圈計數變數，放在外面可相容較舊 C 編譯器 */

    /* 第 2 節：使用 for 迴圈把 scores[0]、scores[1]、scores[2] 加總 */
    for (i = 0; i < 3; i++)
    {
        sum += scores[i];
    }

    /* 第 3 節：除以 3.0，讓結果保留小數 */
    return sum / 3.0;
}

程式 3：practice13_3_main.c

/*
 * ============================================================
 * Practice 13 第 3-2 題：main.c
 * ============================================================
 *
 * 本檔案是主程式。
 * 題目要求：Construct a file "main.c"，使用 average.c 提供的函式，
 * 計算使用者輸入的三科期中考成績平均。
 */

#include <stdio.h>                 /* 第 0 節：printf()、scanf() 需要此標頭檔 */
#include "practice13_3_average.h"  /* 第 0 節：引入自訂標頭檔，才能呼叫 calculate_average() */

int main(void)
{
    /* 第 1 節：宣告變數 */
    int scores[3];   /* 儲存三科成績 */
    double average;  /* 儲存平均分數 */
    int i;           /* 迴圈計數變數 */

    /* 第 2 節：提示使用者輸入資料 */
    printf("Practice 13 第 3 題：計算三科期中考平均分數\n");
    printf("Please input the midterm scores of 3 subjects.\n");

    /* 第 3 節：使用 for 迴圈讀入三科成績 */
    for (i = 0; i < 3; i++)
    {
        printf("Subject %d score: ", i + 1);

        /* scanf() 回傳成功讀取的資料個數；若不是 1，代表輸入錯誤 */
        if (scanf("%d", &scores[i]) != 1)
        {
            printf("Invalid input. Please enter an integer score.\n");
            return 1;
        }
    }

    /* 第 4 節：呼叫 average.c 中定義的 calculate_average() 函式 */
    average = calculate_average(scores);

    /* 第 5 節：輸出平均成績，%.2f 代表顯示到小數點後兩位 */
    printf("The average score is %.2f\n", average);

    return 0;
}

執行結果範例
